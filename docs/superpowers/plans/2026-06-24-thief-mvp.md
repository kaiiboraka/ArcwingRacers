# Thief MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the Thief character as a prefab with smooth grid-based movement, using a pure C# logic layer tested in isolation from Unity.

**Architecture:** `GridMover` is a plain C# class (no Unity lifecycle dependency) owning all movement state and logic. `GridMoverComponent` is a thin MonoBehaviour that ticks `GridMover` in `FixedUpdate` and drives `Rigidbody2D.MovePosition`. `ThiefComponent` is a stub marker. `GridMover` takes raw `float`/`bool` values from the ScriptableObjects rather than SO references directly — keeping it free of Unity serialization for EditMode testability.

**Tech Stack:** Unity 6.3 LTS (6000.3.17f1), C#, Unity Test Framework (NUnit, EditMode), UnityEngine.Physics2D.

## Global Constraints

- Unity 6.3 LTS only — verify APIs against this version
- All scripts under `Assets/_Project/`; never at `Assets/` root
- File header: `// ClassName.cs\n// One-line purpose.\n// Docs: path/to/doc.md`
- Naming: PascalCase classes/methods/public props; `_camelCase` private fields
- `[SerializeField] private` for all inspector-exposed fields — never `public`
- `Rigidbody2D.MovePosition()` in `FixedUpdate` only — never `transform.position` directly
- `Rigidbody2D.interpolation = Interpolate` on all characters
- No new packages without approval

```csharp
using System;
using UnityEngine;

public enum Direction { Up, Down, Left, Right }

public static class DirectionExtensions
{
    public static Direction Opposite(this Direction dir) => dir switch
    {
        Direction.Up    => Direction.Down,
        Direction.Down  => Direction.Up,
        Direction.Left  => Direction.Right,
        Direction.Right => Direction.Left,
        _ => throw new ArgumentOutOfRangeException(nameof(dir), dir, null)
    };

    public static Vector2Int ToVector2Int(this Direction dir) => dir switch
    {
        Direction.Up    => Vector2Int.up,
        Direction.Down  => Vector2Int.down,
        Direction.Left  => Vector2Int.left,
        Direction.Right => Vector2Int.right,
        _ => throw new ArgumentOutOfRangeException(nameof(dir), dir, null)
    };
}
```

```csharp
// GridMover.cs
// Pure C# movement logic for any grid-moving character. No Unity lifecycle dependency.
// Feed Tick(deltaTime) from GridMoverComponent.FixedUpdate.
// Docs: docs/superpowers/specs/2026-06-24-thief-mvp-design.md

using System;
using UnityEngine;

public class GridMover
{
    public Vector2Int CurrentCell      { get; private set; }
    public Vector2Int TargetCell       { get; private set; }
    public Direction  CurrentDirection { get; private set; }
    public float      Progress         { get; private set; }
    public bool       IsMoving         { get; private set; }

    public event Action<Vector2Int> OnCellReached;
    public event Action<Direction>  OnDirectionChanged;
    public event Action             OnStopped;

    private readonly float     _cellSizeWorldUnits;
    private readonly float     _speedCellsPerSecond;
    private readonly bool      _canImmediatelyReverse;
    private readonly IMapQuery _mapQuery;
    private Direction?         _queuedDirection;

    public Vector3 WorldPosition
    {
        get
        {
            var from = new Vector3(CurrentCell.x * _cellSizeWorldUnits,
                                   CurrentCell.y * _cellSizeWorldUnits, 0f);
            var to   = new Vector3(TargetCell.x  * _cellSizeWorldUnits,
                                   TargetCell.y  * _cellSizeWorldUnits, 0f);
            return Vector3.Lerp(from, to, Progress);
        }
    }

    public GridMover(
        float cellSizeWorldUnits,
        float speedCellsPerSecond,
        bool  canImmediatelyReverse,
        Vector2Int startCell,
        Direction  startDirection,
        IMapQuery  mapQuery)
    {
        _cellSizeWorldUnits    = cellSizeWorldUnits;
        _speedCellsPerSecond   = speedCellsPerSecond;
        _canImmediatelyReverse = canImmediatelyReverse;
        _mapQuery              = mapQuery;

        CurrentCell      = startCell;
        CurrentDirection = startDirection;
        TargetCell       = startCell + startDirection.ToVector2Int();
        Progress         = 0f;
        IsMoving         = true;
    }

    public void Tick(float deltaTime)
    {
        if (!IsMoving) return;

        Progress += _speedCellsPerSecond * deltaTime;

        while (Progress >= 1f)
        {
            Progress   -= 1f;
            CurrentCell = TargetCell;
            OnCellReached?.Invoke(CurrentCell);

            Direction  candidate = _queuedDirection ?? CurrentDirection;
            _queuedDirection = null;

            Vector2Int nextCell = CurrentCell + candidate.ToVector2Int();

            if (_mapQuery.IsPassable(nextCell))
            {
                if (candidate != CurrentDirection)
                    OnDirectionChanged?.Invoke(candidate);
                CurrentDirection = candidate;
                TargetCell       = nextCell;
            }
            else if (candidate != CurrentDirection)
            {
                Vector2Int straightCell = CurrentCell + CurrentDirection.ToVector2Int();
                if (_mapQuery.IsPassable(straightCell))
                {
                    TargetCell = straightCell;
                }
                else
                {
                    IsMoving = false;
                    Progress = 0f;
                    OnStopped?.Invoke();
                    return;
                }
            }
            else
            {
                IsMoving = false;
                Progress = 0f;
                OnStopped?.Invoke();
                return;
            }
        }
    }

    public void RequestTurn(Direction direction)
    {
        if (direction == CurrentDirection.Opposite() && _canImmediatelyReverse)
        {
            (CurrentCell, TargetCell) = (TargetCell, CurrentCell);
            CurrentDirection          = direction;
            Progress                  = 1f - Progress;
            OnDirectionChanged?.Invoke(CurrentDirection);
            return;
        }

        if (direction != CurrentDirection.Opposite())
            _queuedDirection = direction;
    }
}
```

```csharp
// Thin MonoBehaviour wrapper: ticks GridMover in FixedUpdate, drives Rigidbody2D.MovePosition.
// Docs: docs/superpowers/specs/2026-06-24-thief-mvp-design.md

using UnityEngine;

[RequireComponent(typeof(Rigidbody2D))]
public class GridMoverComponent : MonoBehaviour
{
    [Header("Config")]
    [SerializeField, Tooltip("Project-wide grid config. Same asset on every character.")]
    private GridConfig _gridConfig;

    [SerializeField, Tooltip("Per-character-type movement settings.")]
    private GridMoverSettings _settings;

    [Header("References")]
    [SerializeField]
    private Rigidbody2D _rigidbody;

    private GridMover _mover;

    public Vector2Int CurrentCell      => _mover.CurrentCell;
    public Direction  CurrentDirection => _mover.CurrentDirection;

    private void Awake()
    {
        _mover = new GridMover(
            cellSizeWorldUnits:    _gridConfig.CellSizeWorldUnits,
            speedCellsPerSecond:   _settings.SpeedCellsPerSecond,
            canImmediatelyReverse: _settings.CanImmediatelyReverse,
            startCell:             Vector2Int.zero,
            startDirection:        Direction.Right,
            mapQuery:              new AlwaysPassableMap());
    }

    private void OnEnable()  => _mover.OnStopped += HandleStopped;
    private void OnDisable() => _mover.OnStopped -= HandleStopped;

    private void FixedUpdate()
    {
        _mover.Tick(Time.fixedDeltaTime);
        _rigidbody.MovePosition(_mover.WorldPosition);
    }

    public void RequestTurn(Direction direction) => _mover.RequestTurn(direction);

    private void HandleStopped()
    {
        // Hook for future wall-bump animation/feedback.
    }

    private void OnValidate()
    {
        if (_gridConfig == null)
            Debug.LogError($"{nameof(GridMoverComponent)} on '{name}': GridConfig not assigned.", this);
        if (_settings == null)
            Debug.LogError($"{nameof(GridMoverComponent)} on '{name}': GridMoverSettings not assigned.", this);
    }
}
```

```csharp
// SetupThiefPrefab.cs
// One-shot editor script: creates GridConfig.asset, ThiefMoverSettings.asset, Thief.prefab.
// Run via: VT → Setup → Create Thief Prefab
// Safe to delete after running, or keep for re-running if assets are lost.

using UnityEditor;
using UnityEngine;

public static class SetupThiefPrefab
{
    private const string ConfigFolder  = "Assets/_Project/ScriptableObjects/Config";
    private const string PrefabFolder  = "Assets/_Project/Prefabs/Characters";
    private const string GridAssetPath = ConfigFolder + "/GridConfig.asset";
    private const string MoverAssetPath = ConfigFolder + "/ThiefMoverSettings.asset";
    private const string PrefabAssetPath = PrefabFolder + "/Thief.prefab";

    [MenuItem("VT/Setup/Create Thief Prefab")]
    public static void Run()
    {
        EnsureFolders();

        var gridConfig    = CreateOrLoad<GridConfig>(GridAssetPath);
        var moverSettings = CreateOrLoad<GridMoverSettings>(MoverAssetPath);

        BuildPrefab(gridConfig, moverSettings);

        AssetDatabase.SaveAssets();
        AssetDatabase.Refresh();

        Debug.Log("Thief setup complete.\n" +
                  $"  GridConfig:         {GridAssetPath}\n" +
                  $"  ThiefMoverSettings: {MoverAssetPath}\n" +
                  $"  Thief prefab:       {PrefabAssetPath}");
    }

    private static void EnsureFolders()
    {
        if (!AssetDatabase.IsValidFolder("Assets/_Project/ScriptableObjects"))
            AssetDatabase.CreateFolder("Assets/_Project", "ScriptableObjects");
        if (!AssetDatabase.IsValidFolder(ConfigFolder))
            AssetDatabase.CreateFolder("Assets/_Project/ScriptableObjects", "Config");
        if (!AssetDatabase.IsValidFolder(PrefabFolder))
            AssetDatabase.CreateFolder("Assets/_Project/Prefabs", "Characters");
    }

    private static T CreateOrLoad<T>(string path) where T : ScriptableObject
    {
        var existing = AssetDatabase.LoadAssetAtPath<T>(path);
        if (existing != null) return existing;
        var asset = ScriptableObject.CreateInstance<T>();
        AssetDatabase.CreateAsset(asset, path);
        return asset;
    }

    private static void BuildPrefab(GridConfig gridConfig, GridMoverSettings moverSettings)
    {
        var go = new GameObject("Thief");

        go.AddComponent<SpriteRenderer>(); // sorting layer set in Project Settings later

        var rb = go.AddComponent<Rigidbody2D>();
        rb.bodyType      = RigidbodyType2D.Kinematic;
        rb.interpolation = RigidbodyInterpolation2D.Interpolate;

        var mover = go.AddComponent<GridMoverComponent>();
        var so    = new SerializedObject(mover);
        so.FindProperty("_gridConfig").objectReferenceValue = gridConfig;
        so.FindProperty("_settings").objectReferenceValue   = moverSettings;
        so.FindProperty("_rigidbody").objectReferenceValue  = rb;
        so.ApplyModifiedPropertiesWithoutUndo();

        go.AddComponent<ThiefComponent>();

        PrefabUtility.SaveAsPrefabAsset(go, PrefabAssetPath);
        Object.DestroyImmediate(go);
    }
}
```
